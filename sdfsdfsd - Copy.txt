```tsx
// Hybrid Visibility Pattern (Desktop hover, Mobile always visible)
<div className="group relative">
  <Card>...</Card>
  
  {/* Actions toolbar */}
  <div className={cn(
    "absolute top-2 right-2 flex gap-1",
    "transition-opacity duration-200",
    "opacity-100",                      // Always visible on mobile
    "md:opacity-0",                     // Hidden on desktop
    "md:group-hover:opacity-100",       // Show on hover (desktop)
    "md:group-focus-within:opacity-100" // Show on keyboard focus

      <Lock className="size-3.5" />
    </Button>

// Touch Long-Press Handler (Gantt)
const useLongPress = (callback: () => void, ms = 500) => {
  const [startLongPress, setStartLongPress] = useState(false)
  
  useEffect(() => {
    let timerId: NodeJS.Timeout
    if (startLongPress) {
      timerId = setTimeout(callback, ms)
    }
    return () => clearTimeout(timerId)
  }, [startLongPress, callback, ms])
  
  return {
    onTouchStart: () => setStartLongPress(true),
    onTouchEnd: () => setStartLongPress(false),
    onMouseDown: () => setStartLongPress(true),
    onMouseUp: () => setStartLongPress(false)
  }
}

// Usage in GanttChart
const longPressHandlers = useLongPress(() => {
  setContextMenu({ x: touchPoint.x, y: touchPoint.y, taskId })
})

<rect {...longPressHandlers} />
```

```tsx
// Table Virtualization Pattern
import { useVirtualizer } from '@tanstack/react-virtual'

const ChangeManagementTable = ({ orders }) => {
  const parentRef = useRef<HTMLDivElement>(null)
  
  const virtualizer = useVirtualizer({
    count: orders.length,
    getScrollElement: () => parentRef.current,
    estimateSize: () => 50,  // Estimated row height
    overscan: 10             // Render 10 extra rows for smooth scroll
  })
  
  return (
    <div ref={parentRef} className="h-[600px] overflow-auto">
      <div
        style={{
          height: `${virtualizer.getTotalSize()}px`,
          width: '100%',
          position: 'relative'
        }}
      >
        {virtualizer.getVirtualItems().map((virtualRow) => (
          <div
            key={virtualRow.key}
            style={{
              position: 'absolute',
              top: 0,
              left: 0,
              width: '100%',
              height: `${virtualRow.size}px`,
              transform: `translateY(${virtualRow.start}px)`
            }}
          >
            <TableRow data={orders[virtualRow.index]} />
          </div>
        ))}
      </div>
    </div>
  )
}

// Grid Virtualization (Documents)
import { useWindowVirtualizer } from '@tanstack/react-virtual'

const DocumentsGrid = ({ documents }) => {
  const virtualizer = useWindowVirtualizer({
    count: documents.length,
    estimateSize: () => 180,  // Card height
    overscan: 5
  })
  
  return (
    <div className="grid grid-cols-1 md:grid-cols-4 gap-4">
      {virtualizer.getVirtualItems().map((virtualItem) => (
        <DocumentCard 
          key={virtualItem.key}
          data={documents[virtualItem.index]}
          style={{
            transform: `translateY(${virtualItem.start}px)`
          }}
        />
      ))}
    </div>
  )
}
```
